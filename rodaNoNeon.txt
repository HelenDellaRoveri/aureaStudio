BEGIN;

-- Verifica se os dados necessários foram cadastrados.

DO $$
DECLARE
  total_clientes INTEGER;
  total_profissionais INTEGER;
  total_servicos INTEGER;
BEGIN
  SELECT COUNT(*)
  INTO total_clientes
  FROM users
  WHERE role = 'client'
    AND email ~ '^cliente[0-9]{2}@aureastudio\.com\.br$';

  SELECT COUNT(*)
  INTO total_profissionais
  FROM professionals
  WHERE id IN (
    'prof-marina-alves',
    'prof-rafael-costa',
    'prof-camila-rocha',
    'prof-bruno-lima',
    'prof-juliana-martins',
    'prof-lucas-ferreira'
  );

  SELECT COUNT(*)
  INTO total_servicos
  FROM services
  WHERE active = TRUE;

  IF total_clientes < 50 THEN
    RAISE EXCEPTION
      'É necessário cadastrar os 50 clientes antes de criar os agendamentos. Clientes encontrados: %',
      total_clientes;
  END IF;

  IF total_profissionais < 6 THEN
    RAISE EXCEPTION
      'É necessário cadastrar os 6 profissionais antes de criar os agendamentos. Profissionais encontrados: %',
      total_profissionais;
  END IF;

  IF total_servicos < 15 THEN
    RAISE EXCEPTION
      'É necessário cadastrar pelo menos 15 serviços. Serviços encontrados: %',
      total_servicos;
  END IF;
END $$;

-- Remove somente os agendamentos criados por este script.
-- Os agendamentos reais não serão apagados.

DELETE FROM appointments
WHERE notes IN (
  'Dados de demonstração - histórico',
  'Dados de demonstração - futuro'
);

-- =========================================================
-- HISTÓRICO DE AGENDAMENTOS
-- 90 registros dos últimos 60 dias
-- =========================================================

WITH client_pool AS (
  SELECT
    id,
    name,
    email,
    ROW_NUMBER() OVER (ORDER BY email) AS position
  FROM users
  WHERE role = 'client'
    AND email ~ '^cliente[0-9]{2}@aureastudio\.com\.br$'
  ORDER BY email
  LIMIT 50
),
professional_pool AS (
  SELECT
    id,
    name,
    service_ids,
    work_days,
    start_time,
    end_time
  FROM professionals
  WHERE id IN (
    'prof-marina-alves',
    'prof-rafael-costa',
    'prof-camila-rocha',
    'prof-bruno-lima',
    'prof-juliana-martins',
    'prof-lucas-ferreira'
  )
    AND active = TRUE
    AND jsonb_array_length(service_ids) > 0
),
historical_candidates AS (
  SELECT
    ROW_NUMBER() OVER (
      ORDER BY generated_day DESC, professional.id
    ) AS sequence_number,
    generated_day::date AS appointment_date,
    professional.id AS professional_id,
    professional.name AS professional_name,
    professional.service_ids,
    professional.start_time
  FROM generate_series(
    GREATEST(
      CURRENT_DATE - 60,
      DATE_TRUNC('year', CURRENT_DATE)::date
    )::timestamp,
    (CURRENT_DATE - 1)::timestamp,
    INTERVAL '1 day'
  ) AS generated_day
  CROSS JOIN professional_pool AS professional
  WHERE professional.work_days @> jsonb_build_array(
    EXTRACT(DOW FROM generated_day)::integer
  )
),
selected_history AS (
  SELECT *
  FROM historical_candidates
  ORDER BY appointment_date DESC, professional_id
  LIMIT 90
),
prepared_history AS (
  SELECT
    client.id AS client_id,
    service.id AS service_id,
    service.name AS service_name,
    selected.professional_id,
    selected.professional_name,
    selected.appointment_date,
    selected.start_time,
    service.duration_minutes,
    service.price_cents,
    CASE
      WHEN selected.sequence_number % 15 = 0 THEN 'Cancelado'
      WHEN selected.sequence_number % 11 = 0 THEN 'Não compareceu'
      ELSE 'Concluído'
    END AS appointment_status,
    selected.sequence_number
  FROM selected_history AS selected
  JOIN client_pool AS client
    ON client.position = (
      ((selected.sequence_number - 1) % 50) + 1
    )
  JOIN services AS service
    ON service.id = (
      selected.service_ids ->> (
        (
          (selected.sequence_number - 1)
          % jsonb_array_length(selected.service_ids)
        )::integer
      )
    )
  WHERE service.active = TRUE
)
INSERT INTO appointments (
  client_id,
  service_id,
  service_name,
  professional_id,
  professional_name,
  appointment_date,
  start_time,
  duration_minutes,
  price_cents,
  status,
  notes,
  created_at,
  updated_at
)
SELECT
  client_id,
  service_id,
  service_name,
  professional_id,
  professional_name,
  appointment_date,
  start_time,
  duration_minutes,
  price_cents,
  appointment_status,
  'Dados de demonstração - histórico',
  appointment_date::timestamp - INTERVAL '5 days',
  appointment_date::timestamp + start_time
FROM prepared_history;

-- =========================================================
-- AGENDAMENTOS FUTUROS
-- 60 registros entre amanhã e o final do ano
-- =========================================================

WITH client_pool AS (
  SELECT
    id,
    name,
    email,
    ROW_NUMBER() OVER (ORDER BY email) AS position
  FROM users
  WHERE role = 'client'
    AND email ~ '^cliente[0-9]{2}@aureastudio\.com\.br$'
  ORDER BY email
  LIMIT 50
),
professional_pool AS (
  SELECT
    id,
    name,
    service_ids,
    work_days,
    start_time,
    end_time
  FROM professionals
  WHERE id IN (
    'prof-marina-alves',
    'prof-rafael-costa',
    'prof-camila-rocha',
    'prof-bruno-lima',
    'prof-juliana-martins',
    'prof-lucas-ferreira'
  )
    AND active = TRUE
    AND jsonb_array_length(service_ids) > 0
),
future_candidates AS (
  SELECT
    ROW_NUMBER() OVER (
      ORDER BY generated_day, professional.id
    ) AS sequence_number,
    generated_day::date AS appointment_date,
    professional.id AS professional_id,
    professional.name AS professional_name,
    professional.service_ids,
    professional.start_time
  FROM generate_series(
    (CURRENT_DATE + 1)::timestamp,
    LEAST(
      CURRENT_DATE + 60,
      MAKE_DATE(
        EXTRACT(YEAR FROM CURRENT_DATE)::integer,
        12,
        31
      )
    )::timestamp,
    INTERVAL '1 day'
  ) AS generated_day
  CROSS JOIN professional_pool AS professional
  WHERE professional.work_days @> jsonb_build_array(
    EXTRACT(DOW FROM generated_day)::integer
  )
),
selected_future AS (
  SELECT *
  FROM future_candidates
  ORDER BY appointment_date, professional_id
  LIMIT 60
),
prepared_future AS (
  SELECT
    client.id AS client_id,
    service.id AS service_id,
    service.name AS service_name,
    selected.professional_id,
    selected.professional_name,
    selected.appointment_date,
    selected.start_time,
    service.duration_minutes,
    service.price_cents,
    CASE
      WHEN selected.sequence_number % 3 = 0 THEN 'Confirmado'
      ELSE 'Agendado'
    END AS appointment_status,
    selected.sequence_number
  FROM selected_future AS selected
  JOIN client_pool AS client
    ON client.position = (
      ((selected.sequence_number - 1) % 50) + 1
    )
  JOIN services AS service
    ON service.id = (
      selected.service_ids ->> (
        (
          (selected.sequence_number - 1)
          % jsonb_array_length(selected.service_ids)
        )::integer
      )
    )
  WHERE service.active = TRUE
)
INSERT INTO appointments (
  client_id,
  service_id,
  service_name,
  professional_id,
  professional_name,
  appointment_date,
  start_time,
  duration_minutes,
  price_cents,
  status,
  notes,
  created_at,
  updated_at
)
SELECT
  client_id,
  service_id,
  service_name,
  professional_id,
  professional_name,
  appointment_date,
  start_time,
  duration_minutes,
  price_cents,
  appointment_status,
  'Dados de demonstração - futuro',
  NOW() - ((sequence_number % 10) || ' days')::interval,
  NOW()
FROM prepared_future;

COMMIT;

-- Exibe o resultado da inserção.

SELECT
  CASE
    WHEN appointment_date < CURRENT_DATE
      THEN 'Histórico'
    ELSE 'Futuro'
  END AS tipo,
  status,
  COUNT(*) AS quantidade,
  ROUND(SUM(price_cents) / 100.0, 2) AS valor_total,
  ROUND(SUM(price_cents) * 0.40 / 100.0, 2) AS parte_salao,
  ROUND(SUM(price_cents) * 0.60 / 100.0, 2) AS parte_profissionais
FROM appointments
WHERE notes IN (
  'Dados de demonstração - histórico',
  'Dados de demonstração - futuro'
)
GROUP BY
  CASE
    WHEN appointment_date < CURRENT_DATE
      THEN 'Histórico'
    ELSE 'Futuro'
  END,
  status
ORDER BY tipo, status;
